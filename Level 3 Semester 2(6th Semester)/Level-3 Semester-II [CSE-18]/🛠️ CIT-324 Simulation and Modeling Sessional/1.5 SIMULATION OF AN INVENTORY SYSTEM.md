import random
import math

# Parameters
INITIAL_INV_LEVEL = 60
NUM_MONTHS = 120
MEAN_INTERDEMAND = 0.1
MIN_LAG = 0.5
MAX_LAG = 1.0
SETUP_COST = 32.0
INCREMENTAL_COST = 3.0
HOLDING_COST = 1.0
SHORTAGE_COST = 5.0
PROB_DISTRIB_DEMAND = [0.0, 0.167, 0.500, 0.833, 1.000]

# State variables
sim_time = 0.0
inv_level = INITIAL_INV_LEVEL
amount = 0
time_last_event = 0.0
total_ordering_cost = 0.0
area_holding = 0.0
area_shortage = 0.0
next_event_type = 0
time_next_event = [0, float('inf'), 0, NUM_MONTHS, 0]
num_events = 4
# Random variate generation (exponential)
def expon(mean):
    u = random.random()
    return -mean * math.log(u)

# Random integer generation (discrete)
def random_integer(prob_distrib):
    u = random.random()
    for i in range(1, len(prob_distrib)):
        if u <= prob_distrib[i]:
            return i
    return len(prob_distrib) - 1

# Random variate generation (uniform)
def uniform(a, b):
    return a + random.random() * (b - a)

# Event list management (using a simple list)
def timing():
    min_time_next_event = float('inf')
    next_event_type = 0
    for i in range(1,num_events + 1):
      if(time_next_event[i] < min_time_next_event):
        min_time_next_event = time_next_event[i]
        next_event_type = i
    if (next_event_type == 0):
        print(f"\nEvent list empty at time {sim_time}")
        exit()
    sim_time = min_time_next_event
    return next_event_type

def update_time_avg_stats():
  global area_holding, area_shortage, time_last_event
  time_since_last_event = sim_time - time_last_event
  if inv_level > 0:
    area_holding += inv_level * time_since_last_event
  elif inv_level < 0:
    area_shortage += abs(inv_level) * time_since_last_event
  time_last_event = sim_time

# Event routines
def order_arrival():
    global inv_level, time_next_event
    inv_level += amount
    time_next_event = float('inf')

def demand():
  global inv_level, time_next_event
  inv_level -= random_integer(PROB_DISTRIB_DEMAND)
  time_next_event = sim_time + expon(MEAN_INTERDEMAND)

def evaluate(smalls,bigs):
    global total_ordering_cost, amount, time_next_event
    if inv_level < smalls:

      total_ordering_cost += SETUP_COST + INCREMENTAL_COST * amount
      time_next_event = sim_time + uniform(MIN_LAG, MAX_LAG)
    time_next_event = sim_time + 1.0

def report(smalls, bigs):

        order_arrival()
      elif next_event == 2:

      elif next_event == 4:

    report(smalls, bigs)
